#!/usr/bin/env python3
"""
HIcosmo High-Level API - Minimal Interface for Cosmological Parameter Estimation.

This module provides the `hicosmo()` factory function for zero-configuration
parameter inference with automatic parameter matching, multi-likelihood support,
and intelligent defaults.

Examples
--------
>>> from hicosmo import hicosmo
>>> from hicosmo.models import LCDM
>>> from hicosmo.likelihoods import SN_likelihood
>>>
>>> # Single likelihood
>>> inference = hicosmo(
...     cosmology=LCDM,
...     likelihood=SN_likelihood(LCDM, "pantheon+"),
...     free_params=['H0', 'Omega_m']
... )
>>> samples = inference.run(num_samples=8000)
>>>
>>> # Multiple likelihoods
>>> inference = hicosmo(
...     cosmology=LCDM,
...     likelihood=[pantheon, bao, cmb_prior],
...     free_params=['H0', 'Omega_m']
... )
>>> samples = inference.run(num_samples=8000)
>>> inference.summary()
>>> inference.corner_plot('results/corner.pdf')
"""

from typing import Type, Union, List, Callable, Dict, Any, Optional
import warnings
import inspect
from pathlib import Path

from .models.base import CosmologyBase
from .parameters import ParameterRegistry
from .samplers import MCMC, ParameterConfig
from .utils.logging import get_logger

logger = get_logger(__name__)

# String-to-class mapping for convenience
COSMOLOGY_MAP = {}
LIKELIHOOD_MAP = {}


def _initialize_maps():
    """Lazy initialize cosmology and likelihood maps to avoid circular imports."""
    global COSMOLOGY_MAP, LIKELIHOOD_MAP

    if not COSMOLOGY_MAP:
        from .models import LCDM
        try:
            from .models import wCDM, CPL
            COSMOLOGY_MAP = {
                'LCDM': LCDM,
                'lcdm': LCDM,
                'wCDM': wCDM,
                'wcdm': wCDM,
                'CPL': CPL,
                'cpl': CPL,
            }
        except ImportError:
            COSMOLOGY_MAP = {
                'LCDM': LCDM,
                'lcdm': LCDM,
            }

    if not LIKELIHOOD_MAP:
        from .likelihoods import (
            SN_likelihood,
            BAO_likelihood,
            H0LiCOWLikelihood,
            Planck2018DistancePriorsLikelihood,
            SH0ESLikelihood,
            TDCOSMOLikelihood,
        )
        LIKELIHOOD_MAP = {
            # Supernovae
            'sn': lambda: SN_likelihood(dataset="pantheon+", M_B="marginalize", include_systematics=True),
            'sn_shoes': lambda: SN_likelihood(dataset="pantheon+shoes", M_B="marginalize", include_systematics=True),

            # BAO datasets
            'bao': lambda: BAO_likelihood(dataset="desi2024"),
            'bao_sdss_dr12': lambda: BAO_likelihood(dataset="sdss_dr12"),
            'bao_sdss_dr16': lambda: BAO_likelihood(dataset="sdss_dr16"),
            'bao_boss_dr12': lambda: BAO_likelihood(dataset="boss_dr12"),
            'bao_sixdf': lambda: BAO_likelihood(dataset="sixdf"),

            # Strong lensing time delays
            'h0licow': lambda: H0LiCOWLikelihood(),
            'tdcosmo': lambda: TDCOSMOLikelihood(),

            # CMB distance priors
            'planck': lambda: Planck2018DistancePriorsLikelihood(),
            'planck2018': lambda: Planck2018DistancePriorsLikelihood(),
            'cmb': lambda: Planck2018DistancePriorsLikelihood(),

            # Direct H0 measurements
            'shoes': lambda: SH0ESLikelihood(),
            'sh0es': lambda: SH0ESLikelihood(),
            'h0': lambda: SH0ESLikelihood(),
        }

class InferenceRunner:
    """
    High-level inference execution wrapper.

    This class encapsulates the complete MCMC workflow with automatic parameter
    matching, likelihood combination, and cosmology model instantiation.

    Users should create instances via the `hicosmo()` factory function rather
    than directly instantiating this class.

    Attributes
    ----------
    cosmology_class : Type[CosmologyBase]
        Cosmology model class (e.g., LCDM, wCDM).
    likelihoods : List
        List of likelihood objects.
    registry : ParameterRegistry
        Parameter registry with free/fixed parameter management.
    mcmc : MCMC
        Underlying MCMC sampler instance.
    """

    def __init__(
        self,
        cosmology_class: Type[CosmologyBase],
        likelihoods: List,
        registry: ParameterRegistry,
        mcmc_config: Dict[str, Any],
        setup_mcmc: bool = True,
    ):
        """
        Initialize inference runner.

        Parameters
        ----------
        cosmology_class : Type[CosmologyBase]
            Cosmology model class (not instance).
        likelihoods : List
            List of likelihood objects (already instantiated).
        registry : ParameterRegistry
            Parameter registry with configured free/fixed parameters.
        mcmc_config : Dict
            MCMC configuration dictionary.
        """
        self.cosmology_class = cosmology_class
        self.likelihoods = likelihoods
        self.registry = registry
        self.mcmc_config = mcmc_config

        # Create combined likelihood function
        self.likelihood_func = self._create_combined_likelihood()

        # Setup MCMC sampler
        if setup_mcmc:
            self._setup_mcmc()



    def _create_combined_likelihood(self) -> Callable:
        """
        Create combined log-likelihood function with explicit parameter signature.

        This creates a function that:
        1. Has explicit parameters matching free parameters
        2. Accepts fixed parameters as additional keyword arguments
        3. Dynamically instantiates the cosmology model
        4. Calls each likelihood and sums log-likelihoods

        Returns
        -------
        Callable
            Combined log-likelihood function with explicit signature.
        """
        # Get free and fixed parameter names
        free_param_names = list(self.registry.get_free().keys())
        fixed_param_values = {name: param.value
                             for name, param in self.registry.get_fixed().items()}

        # Create a closure that captures the context
        cosmology_class = self.cosmology_class
        likelihoods = self.likelihoods
        call_likelihood_smart = self._call_likelihood_smart

        # Create function dynamically with explicit parameters
        # The explicit signature enables direct parameter mapping in MCMC
        func_code = f"""
def combined_loglike({', '.join(free_param_names)}):
    '''Combined log-likelihood for {', '.join(free_param_names)}'''
    # Merge free and fixed parameters
    all_params = {{**fixed_params, **{{{', '.join(f"'{p}': {p}" for p in free_param_names)}}}}}

    # Create cosmology model instance
    try:
        model = cosmology_class(**all_params)
    except TypeError:
        # Filter parameters to match model signature
        sig = inspect.signature(cosmology_class.__init__)
        valid_params = {{k: v for k, v in all_params.items()
                       if k in sig.parameters and k != 'self'}}
        model = cosmology_class(**valid_params)

    # Compute combined log-likelihood
    total_loglike = 0.0
    for likelihood in likelihoods:
        loglike = call_likelihood_smart(likelihood, model, all_params)
        total_loglike += loglike

    return total_loglike
"""

        # Execute the function definition
        local_vars = {
            'fixed_params': fixed_param_values,
            'cosmology_class': cosmology_class,
            'likelihoods': likelihoods,
            'call_likelihood_smart': call_likelihood_smart,
            'inspect': inspect
        }
        exec(func_code, local_vars)

        return local_vars['combined_loglike']

    def _call_likelihood_smart(
        self,
        likelihood,
        model: CosmologyBase,
        all_params: Dict[str, float]
    ) -> float:
        """
        Intelligently call likelihood with correct arguments.

        Prefers __call__(**params) for JAX-traced fast path, falls back to
        log_likelihood(model) for legacy compatibility.

        This handles different likelihood signatures:
        - __call__(**params) - preferred, uses JAX-traced fast path
        - log_likelihood(model)
        - log_likelihood(cosmology)
        - log_likelihood(model, M_B=...)

        Parameters
        ----------
        likelihood : Likelihood
            Likelihood object.
        model : CosmologyBase
            Cosmology model instance.
        all_params : Dict[str, float]
            All parameters (for nuisance parameters like M_B).

        Returns
        -------
        float
            Log-likelihood value.
        """
        # PRIORITY 1: Try likelihood.__call__(**params) for JAX-traced fast path
        # This is the preferred path for MCMC with gradient-based samplers
        if hasattr(likelihood, '__call__') and callable(likelihood):
            try:
                # Call likelihood object directly with params (uses fast path)
                return likelihood(**all_params)
            except (TypeError, KeyError):
                # Fall through to slow path if __call__ signature doesn't match
                pass

        # PRIORITY 2: Try log_likelihood with model (slow path, but JAX compatible if traced)
        try:
            sig = inspect.signature(likelihood.log_likelihood)
            param_names = list(sig.parameters.keys())

            # Build keyword arguments
            kwargs = {}

            # Check first parameter name (model vs cosmology)
            if len(param_names) > 0:
                first_param = param_names[0]
                if first_param in ['model', 'cosmology']:
                    kwargs[first_param] = model

            # Add any nuisance parameters that the likelihood expects
            for param_name in param_names[1:]:  # Skip first parameter (model/cosmology)
                if param_name in all_params:
                    kwargs[param_name] = all_params[param_name]

            # If the likelihood accepts **kwargs, pass all remaining parameters
            if any(param.kind == inspect.Parameter.VAR_KEYWORD for param in sig.parameters.values()):
                for key, value in all_params.items():
                    if key not in kwargs:
                        kwargs[key] = value

            # Call with keyword arguments if we matched any, otherwise positional
            if kwargs:
                return likelihood.log_likelihood(**kwargs)
            else:
                return likelihood.log_likelihood(model)

        except Exception as e:
            # Fallback: try positional argument
            try:
                return likelihood.log_likelihood(model)
            except Exception as e2:
                raise RuntimeError(
                    f"Failed to call {likelihood.__class__.__name__}.log_likelihood. "
                    f"Tried __call__(**params), keyword args, and positional. "
                    f"Errors: {str(e)}, {str(e2)}"
                ) from e2

    def _setup_mcmc(self):
        """
        Setup MCMC sampler with automatic parameter configuration.

        Converts ParameterRegistry to ParameterConfig format expected by MCMC.
        """
        # Build parameter dictionary for ParameterConfig
        param_dict = {}
        for name, param in self.registry.get_free().items():
            # Convert to tuple format: (ref_value, min, max)
            if param.prior and param.prior.get('dist') == 'uniform':
                param_dict[name] = (
                    param.value,
                    param.prior['min'],
                    param.prior['max']
                )
            elif param.bounds:
                param_dict[name] = (
                    param.value,
                    param.bounds[0],
                    param.bounds[1]
                )
            else:
                # Default wide range if no prior/bounds specified
                param_dict[name] = (param.value, param.value * 0.5, param.value * 1.5)

        # Split sampler selection from other MCMC kwargs
        sampler_name = self.mcmc_config.get('sampler', 'numpyro')
        mcmc_options = {k: v for k, v in self.mcmc_config.items() if k not in {'sampler', 'derived'}}

        # Create ParameterConfig
        derived = self.mcmc_config.get("derived", [])
        config = ParameterConfig(
            param_dict,
            derived=derived,
            mcmc=mcmc_options
        )

        # Provide deterministic initial values for gradient-based samplers
        if sampler_name == 'numpyro':
            init_values = {
                name: param.value
                for name, param in self.registry.get_free().items()
                if param.value is not None
            }
            config.mcmc.setdefault('init_strategy', 'value')
            config.mcmc.setdefault('init_values', init_values)

        # Create MCMC instance
        # Note: No need to pass fixed_params as data_kwargs because they are
        # already captured in the likelihood_func closure
        chain_name = self.mcmc_config.get('chain_name')
        self.mcmc = MCMC(config, self.likelihood_func, sampler=sampler_name, chain_name=chain_name)

    def run(self, num_samples: Optional[int] = None, num_chains: Optional[int] = None, **kwargs):
        """
        Run MCMC sampling.

        Parameters
        ----------
        num_samples : int, optional
            Total number of samples (across all chains).
            Will be divided by num_chains to get samples per chain.
            If not specified, uses value from hicosmo() initialization.
        num_chains : int, optional
            Number of parallel chains.
            If not specified, uses value from hicosmo() initialization.
        **kwargs
            Additional MCMC parameters (e.g., num_warmup, target_accept).

        Returns
        -------
        dict
            MCMC samples dictionary with parameter names as keys.
        """
        # Get current values
        current_samples = num_samples or self.mcmc.param_config.mcmc.get('num_samples', 8000)
        current_chains = num_chains or self.mcmc.param_config.mcmc.get('num_chains', 4)

        # Update MCMC configuration with TOTAL samples (across all chains).
        # MCMC._apply_intelligent_defaults will convert totals to per-chain values.
        self.mcmc.param_config.mcmc['num_samples'] = current_samples
        self.mcmc.param_config.mcmc['num_chains'] = current_chains

        # Update additional MCMC parameters
        self.mcmc.param_config.mcmc.update(kwargs)

        # Run MCMC sampling
        return self.mcmc.run()

def hicosmo(
    cosmology: Union[Type[CosmologyBase], str],
    likelihood: Union[Any, List[Any], str],
    free_params: List[str],
    preset: str = 'planck2018',
    num_samples: int = 8000,
    num_chains: int = 4,
    strict_mode: bool = False,
    **mcmc_kwargs
) -> InferenceRunner:
    """
    Create cosmological parameter inference with zero configuration.

    This is the main high-level API for HIcosmo. It automatically:
    - Loads parameter defaults from presets (Planck 2018, WMAP9, etc.)
    - Sets specified parameters as free, others as fixed
    - Handles nuisance parameters (e.g., M_B marginalization)
    - Combines multiple likelihoods
    - Configures MCMC sampling

    Parameters
    ----------
    cosmology : Type[CosmologyBase] or str
        Cosmology model CLASS (not instance) or string name.
        Examples: LCDM, wCDM, CPL or 'LCDM', 'wCDM', 'CPL'
    likelihood : Likelihood, List[Likelihood], or str
        Single likelihood object/string or list of likelihood objects/strings.
        Supported string shortcuts:
        - Supernovae: 'sn', 'sn_shoes'
        - BAO: 'bao', 'bao_sdss_dr12', 'bao_sdss_dr16', 'bao_boss_dr12', 'bao_sixdf'
        - Strong Lensing: 'h0licow', 'tdcosmo'
        - CMB: 'planck', 'planck2018', 'cmb'
        - H0: 'shoes', 'sh0es', 'h0'
        Each likelihood must have a log_likelihood() method.
    free_params : List[str]
        List of parameter names to vary in MCMC sampling.
        Example: ['H0', 'Omega_m']
    preset : str, default='planck2018'
        Parameter preset name for default values.
        Options: 'planck2018', 'wmap9'
    num_samples : int, default=8000
        Total number of MCMC samples (across all chains).
    num_chains : int, default=4
        Number of parallel MCMC chains.
    strict_mode : bool, default=False
        If True, raise errors for parameter mismatches.
        If False, issue warnings and attempt auto-correction.
    **mcmc_kwargs
        Additional MCMC configuration parameters.
        Examples: num_warmup, target_accept, max_tree_depth

    Returns
    -------
    InferenceRunner
        Inference execution object. Call .run() to execute sampling.

    Examples
    --------
    >>> # Ultra-simple string-based inference
    >>> from hicosmo import hicosmo
    >>>
>>> inference = hicosmo(
...     cosmology='LCDM',
...     likelihood='sn',
...     free_params=['H0', 'Omega_m']
... )
    >>> samples = inference.run()

    >>> # Traditional class-based inference
    >>> from hicosmo.models import LCDM
>>> from hicosmo.likelihoods import SN_likelihood
    >>>
>>> inference = hicosmo(
...     cosmology=LCDM,
...     likelihood=SN_likelihood(LCDM, "pantheon+"),
...     free_params=['H0', 'Omega_m']
... )
    >>> samples = inference.run()

    >>> # Multi-probe joint analysis
>>> inference = hicosmo(
...     cosmology='LCDM',
...     likelihood=['sn_shoes', 'bao', 'cmb'],
...     free_params=['H0', 'Omega_m'],
...     num_samples=16000,
...     num_chains=8
... )
    >>> samples = inference.run()
    """
    # Step 0: Initialize string-to-class mappings
    _initialize_maps()

    # Step 1: Resolve cosmology (string or class)
    if isinstance(cosmology, str):
        if cosmology not in COSMOLOGY_MAP:
            raise ValueError(
                f"Unknown cosmology model '{cosmology}'. "
                f"Available models: {list(COSMOLOGY_MAP.keys())}"
            )
        cosmology_class = COSMOLOGY_MAP[cosmology]
    else:
        cosmology_class = cosmology

    # Step 2: Normalize and resolve likelihood(s)
    if not isinstance(likelihood, list):
        likelihood_list = [likelihood]
    else:
        likelihood_list = likelihood

    likelihoods = []
    for like in likelihood_list:
        if isinstance(like, str):
            if like not in LIKELIHOOD_MAP:
                raise ValueError(
                    f"Unknown likelihood '{like}'. "
                    f"Available likelihoods: {list(LIKELIHOOD_MAP.keys())}"
                )
            likelihoods.append(LIKELIHOOD_MAP[like]())  # Call lambda to create instance
        else:
            likelihoods.append(like)

    # Step 2.5: Set cosmology_class on likelihoods for JAX-traced fast path
    # This enables likelihoods to use compute_grid_traced() instead of slow model instantiation
    for like in likelihoods:
        if hasattr(like, 'cosmology_class') and like.cosmology_class is None:
            like.cosmology_class = cosmology_class
            # Re-initialize fast likelihood if the method exists
            if hasattr(like, '_initialize_fast_likelihood'):
                like._initialize_fast_likelihood()

    # Step 3: Create parameter registry from preset
    try:
        registry = ParameterRegistry.from_defaults(preset)
    except Exception as e:
        raise ValueError(
            f"Failed to load preset '{preset}'. "
            f"Available presets: planck2018, wmap9. "
            f"Error: {str(e)}"
        ) from e

    # Step 3.5: Merge cosmology-model parameters (e.g., w for wCDM)
    try:
        from .parameters import Parameter

        get_params = getattr(cosmology_class, "get_parameters", None)
        if callable(get_params):
            for param in get_params():
                if not isinstance(param, Parameter):
                    continue
                if param.name in registry:
                    continue
                registry.add(
                    param.name,
                    value=param.value,
                    free=param.free,
                    prior=param.prior,
                    latex_label=param.latex_label,
                    description=param.description,
                    bounds=param.bounds,
                )
    except Exception:
        # Non-fatal: proceed with preset-only parameters
        pass

    # Step 4: Handle nuisance parameters from likelihoods
    for like in likelihoods:
        before = set(registry.get_all().keys())
        try:
            registry.add_from_likelihood(like)
        except Exception:
            continue
        added = set(registry.get_all().keys()) - before
        for name in added:
            try:
                if registry.get(name).free and name not in free_params:
                    free_params = free_params + [name]
            except Exception:
                continue

    # Step 5: Set free parameters
    try:
        registry.set_free(free_params)
    except KeyError as e:
        if strict_mode:
            raise ValueError(
                f"Parameter {str(e)} not found in preset '{preset}'. "
                f"Available parameters: {list(registry.get_all().keys())}"
            ) from e
        else:
            # Try to add missing parameter with default prior
            missing_param = str(e).split("'")[1] if "'" in str(e) else str(e)
            raise ValueError(
                f"Parameter '{missing_param}' not in preset '{preset}'. "
                f"Available parameters: {list(registry.get_all().keys())}"
            ) from e

    # Step 6: Validate cosmology class accepts these parameters (silently)
    # Suppress warnings about unused parameters - this is expected behavior
    with warnings.catch_warnings():
        warnings.filterwarnings('ignore', category=UserWarning)
        try:
            sig = inspect.signature(cosmology_class.__init__)
            cosmo_params = set(sig.parameters.keys()) - {'self', 'kwargs', 'args'}
            registry_params = set(registry.get_all().keys())
        except Exception:
            # If we can't inspect, just proceed
            pass

    # Step 7: Configure MCMC
    mcmc_config = {
        'num_samples': num_samples,
        'num_chains': num_chains,
        'verbose': mcmc_kwargs.pop('verbose', True),
        **mcmc_kwargs
    }

    # Step 8: Create and return InferenceRunner
    return InferenceRunner(
        cosmology_class=cosmology_class,
        likelihoods=likelihoods,
        registry=registry,
        mcmc_config=mcmc_config
    )

def list_likelihoods(verbose: bool = True) -> Dict[str, str]:
    """
    List all supported likelihood strings for the hicosmo() API.

    Automatically retrieves from LIKELIHOOD_MAP - no hardcoding.

    Parameters
    ----------
    verbose : bool, default=True
        If True, print formatted output to console.

    Returns
    -------
    Dict[str, str]
        Dictionary mapping likelihood string to description.
    """
    _initialize_maps()

    # Extract class name and parameters from lambda source
    import re
    result = {}

    for key, factory in LIKELIHOOD_MAP.items():
        try:
            source = inspect.getsource(factory)
            # Extract full lambda expression: "lambda: ClassName(...)"
            match = re.search(r"'[^']+'\s*:\s*lambda:\s*(\w+)\s*\(([^)]*)\)", source)
            if match:
                cls_name = match.group(1)
                params = match.group(2).strip()
                if params:
                    result[key] = f"{cls_name}({params})"
                else:
                    result[key] = f"{cls_name}()"
            else:
                # Fallback: just get class name
                match2 = re.search(r'lambda:\s*(\w+)\s*\(', source)
                if match2:
                    result[key] = match2.group(1)
                else:
                    result[key] = "Unknown"
        except Exception:
            result[key] = "Unknown"

    if verbose:
        logger.info("\n" + "=" * 70)
        logger.info("HIcosmo Supported Likelihoods")
        logger.info("=" * 70)

        # Group by category based on class name patterns
        categories = {
            'Supernovae': [],
            'BAO': [],
            'Strong Lensing': [],
            'CMB': [],
            'H0 Direct': [],
        }

        for key, desc in result.items():
            if 'SN_likelihood' in desc or 'Pantheon' in desc:
                categories['Supernovae'].append((key, desc))
            elif 'BAO_likelihood' in desc or 'BAO' in desc or 'SixDF' in desc:
                categories['BAO'].append((key, desc))
            elif 'H0LiCOW' in desc or 'TDCOSMO' in desc:
                categories['Strong Lensing'].append((key, desc))
            elif 'Planck' in desc:
                categories['CMB'].append((key, desc))
            elif 'SH0ES' in desc:
                categories['H0 Direct'].append((key, desc))

        for category, items in categories.items():
            if items:
                logger.info(f"\n{category}:")
                for key, desc in items:
                    logger.info(f"  '{key:18s} -> {desc}")

        logger.info("\n" + "-" * 70)
        logger.info("Usage:")
        logger.info("  hicosmo('LCDM', 'bao', ['H0', 'Omega_m'])")
        logger.info("  hicosmo('LCDM', ['sn', 'bao', 'cmb'], ['H0', 'Omega_m'])")
        logger.info("=" * 70 + "\n")

    return result

def list_cosmologies(verbose: bool = True) -> Dict[str, Dict[str, Any]]:
    """
    List all supported cosmology models and their parameters.

    Automatically retrieves from COSMOLOGY_MAP and inspects class signatures.

    Parameters
    ----------
    verbose : bool, default=True
        If True, print formatted output to console.

    Returns
    -------
    Dict[str, Dict[str, Any]]
        Dictionary mapping model names to their parameter information.
    """
    _initialize_maps()

    # Auto-detect: group by class to find aliases
    class_to_keys = {}
    for key, cls in COSMOLOGY_MAP.items():
        cls_name = cls.__name__
        if cls_name not in class_to_keys:
            class_to_keys[cls_name] = {'class': cls, 'aliases': []}
        class_to_keys[cls_name]['aliases'].append(key)

    # Build result by inspecting each class
    result = {}
    for cls_name, info in class_to_keys.items():
        cls = info['class']
        aliases = info['aliases']

        # Get parameters from __init__ signature
        try:
            sig = inspect.signature(cls.__init__)
            params = {}
            for param_name, param in sig.parameters.items():
                if param_name in ('self', 'args', 'kwargs'):
                    continue
                default = param.default if param.default != inspect.Parameter.empty else None
                params[param_name] = default
        except Exception:
            params = {}

        # Get docstring
        doc = cls.__doc__.strip().split('\n')[0] if cls.__doc__ else "No description"

        result[cls_name] = {
            'class': cls,
            'aliases': aliases,
            'parameters': params,
            'description': doc,
        }

    if verbose:
        logger.info("\n" + "=" * 60)
        logger.info("HIcosmo Supported Cosmology Models")
        logger.info("=" * 60)

        for cls_name, info in result.items():
            logger.info(f"\n{cls_name}")
            logger.info(f"  {info['description']}")
            logger.info(f"  Strings: {', '.join(repr(a) for a in info['aliases'])}")
            logger.info("  Parameters (with defaults):")
            for param, default in info['parameters'].items():
                if default is not None:
                    logger.info(f"    {param:12s} = {default}")
                else:
                    logger.info(f"    {param:12s} (required)")

        logger.info("\n" + "-" * 60)
        logger.info("Usage:")
        logger.info("  hicosmo('LCDM', 'bao', ['H0', 'Omega_m'])")
        logger.info("  hicosmo('wCDM', 'sn', ['H0', 'Omega_m', 'w0'])")
        logger.info("=" * 60 + "\n")

    return result

#!/usr/bin/env python3
"""
HIcosmo Unified Plotter Class
=============================

Single entry point for all visualization needs:
- MCMC chain visualization (corner plots, traces, marginals)
- Multi-chain comparison
- Fisher forecast contours

Core Usage (3 lines):
```python
from hicosmo.visualization import Plotter

plotter = Plotter('results/mcmc_chain.pkl')
plotter.corner(['H0', 'Omega_m'], filename='corner.pdf')
```

Author: Jingzhao Qi
"""

from typing import Dict, List, Optional, Union, Any, Tuple

    HAS_GETDIST = False

class Plotter:
    """
    HIcosmo unified visualization interface.

    Supports multiple data input formats:
    - Chain name: 'my_chain' (auto-loads from mcmc_chains/my_chain.h5)
    - Chain names list: ['chain1', 'chain2'] (multi-chain mode)
    - Dict: {'H0': arr, 'Omega_m': arr}
    - NumPy array: np.array((N, D))
    - MCSamples: GetDist native object

    Examples
    --------
    >>> # Single chain (simplest!)
    >>> plotter = Plotter('my_chain')
    >>> plotter.corner(['H0', 'Omega_m'], filename='corner.pdf')

    >>> # Multi-chain comparison
    >>> plotter = Plotter(
    ...     ['low_noise', 'high_noise'],
    ...     labels=['Low Noise', 'High Noise']
    ... )
    >>> plotter.corner(['a', 'b', 'c'], filename='comparison.pdf')

    >>> # From dict
    >>> plotter = Plotter({'H0': samples_h0, 'Omega_m': samples_om})

    >>> # Fisher forecast
    >>> plotter = Plotter.from_fisher(mean, cov, param_names=['w0', 'wa'])
    """

    def __init__(
        self,
        data: Union[str, Path, np.ndarray, dict, List, MCSamples],
        *,
        labels: Optional[Union[List[str], Dict[str, str]]] = None,
        ranges: Optional[Dict[str, Tuple[float, float]]] = None,
        chain_labels: Optional[List[str]] = None,
        style: str = 'modern',
        results_dir: Union[str, Path] = 'results',
        **kwargs,  # Accept but ignore deprecated parameters
    ):
        """
        Initialize Plotter with data.

        Parameters
        ----------
        data : various
            Chain data in various formats (see class docstring)
        labels : list or dict, optional
            Parameter LaTeX labels. If None, auto-extracted from file metadata
            or auto-generated.
        ranges : dict, optional
            Parameter plotting ranges {name: (min, max)}. If None, auto-extracted
            from file metadata or computed from samples.
        chain_labels : list of str, optional
            Labels for multi-chain mode ['Planck', 'SH0ES', ...]
        style : str
            Color scheme ('modern' or 'classic')
        results_dir : str or Path
            Directory for saving plots

        Notes
        -----
        Derived parameters (rd, rd_h, H0_rd, etc.) are now computed automatically
        during MCMC sampling and saved with the chain. No need to pass cosmology.

        Examples
        --------
        >>> # Derived parameters are automatically available from saved chain
        >>> plotter = Plotter('my_chain')
        >>> plotter.corner(['Omega_m', 'rd_h'])  # rd_h from MCMC samples
        """
        _require_getdist()
        # Warn about deprecated cosmology parameter
        if 'cosmology' in kwargs:
            import warnings
            warnings.warn(
                "The 'cosmology' parameter is deprecated and will be ignored. "
                "Derived parameters are now computed during MCMC sampling.",
                DeprecationWarning,
                stacklevel=2
            )

        self.style = style
        self.results_dir = Path(results_dir)
        self.results_dir.mkdir(exist_ok=True)

        # Smart label detection: when data is List[str] (chain names) and labels
        # is List[str], treat labels as chain_labels for legend (user-friendly API)
        is_multichain_names = (
            isinstance(data, (list, tuple)) and len(data) > 0 and
            all(isinstance(item, str) and not Path(item).suffix for item in data)
        )
        if is_multichain_names and isinstance(labels, list) and all(isinstance(l, str) for l in labels):
            # labels is for legend, not parameter LaTeX labels
            chain_labels = labels
            labels = None

        # Store user-provided metadata
        self._user_labels = labels
        self._user_ranges = ranges
        self._chain_labels = chain_labels

        # File metadata (populated if loading from file)
        self._file_metadata: Dict[str, Any] = {}

        # Process data
        self._samples_list: List[MCSamples] = []
        self._is_multichain = False
        self._param_names: List[str] = []
        self._raw_samples: Optional[dict] = None  # Store raw samples for derived param computation

        self._load_data(data)

    def _load_data(self, data):
        """Process input data into MCSamples list.

        Supports:
        - str: chain_name, loads from mcmc_chains/{name}.h5
        - List[str]: multiple chain_names, multi-chain mode
        - dict: {'param': samples}
        - np.ndarray: raw samples
        - MCSamples: GetDist native object
        """
        import h5py

        # Check if it's a chain_name string (not a file path)
        if isinstance(data, str) and not Path(data).suffix:
            # It's a chain_name, load from default directory
            samples_dict, file_ranges, file_labels = self._load_h5_chain(data)
            # Use file ranges if user didn't provide ranges
            if not self._user_ranges and file_ranges:
                self._user_ranges = file_ranges
            # Use file labels if user didn't provide labels
            if not self._user_labels and file_labels:
                self._user_labels = file_labels
            # Derived parameters are already in samples from MCMC
            self._raw_samples = samples_dict
            data = samples_dict

        # Check for list of chain_names (all strings without extensions)
        if isinstance(data, (list, tuple)) and len(data) > 0:
            if all(isinstance(item, str) and not Path(item).suffix for item in data):
                # All items are chain_names
                self._is_multichain = True
                loaded_data = [self._load_h5_chain(name) for name in data]
                all_samples_data = [d[0] for d in loaded_data]  # Extract samples dicts
                all_file_ranges = [d[1] for d in loaded_data]   # Extract ranges dicts
                all_file_labels = [d[2] for d in loaded_data]   # Extract labels dicts

                # Use file ranges from first chain if user didn't provide
                # (all chains should have same prior ranges)
                if self._user_ranges:
                    effective_ranges = self._user_ranges
                elif all_file_ranges[0]:
                    effective_ranges = all_file_ranges[0]
                else:
                    effective_ranges = self._compute_unified_ranges(all_samples_data)

                # Use file labels from first chain if user didn't provide
                effective_labels = self._user_labels if self._user_labels else all_file_labels[0]

                for i, chain_data in enumerate(all_samples_data):
                    samples = _prepare_getdist_samples(
                        chain_data,
                        labels=effective_labels,
                        ranges=effective_ranges,
                    )
                    if self._chain_labels and i < len(self._chain_labels):
                        samples.label = self._chain_labels[i]
                    else:
                        samples.label = f'Chain {i+1}'
                    self._samples_list.append(samples)

                self._param_names = self._samples_list[0].getParamNames().list()
                return

        # Check for multi-chain input (list of dicts/arrays)
        if isinstance(data, (list, tuple)) and len(data) > 1:
            if not (len(data) == 2 and isinstance(data[1], dict) and
                    ('labels' in data[1] or 'ranges' in data[1])):
                self._is_multichain = True

                all_samples_data = []
                for chain_data in data:
                    if isinstance(chain_data, dict):
                        all_samples_data.append(chain_data)
                    elif isinstance(chain_data, np.ndarray):
                        all_samples_data.append(chain_data)

                unified_ranges = self._compute_unified_ranges(all_samples_data)
                effective_ranges = self._user_ranges if self._user_ranges else unified_ranges

                for i, chain_data in enumerate(data):
                    samples = _prepare_getdist_samples(
                        chain_data,
                        labels=self._user_labels,
                        ranges=effective_ranges,
                    )
                    if self._chain_labels and i < len(self._chain_labels):
                        samples.label = self._chain_labels[i]
                    else:
                        samples.label = f'Chain {i+1}'
                    self._samples_list.append(samples)

                self._param_names = self._samples_list[0].getParamNames().list()
                return

        # Single chain mode
        if isinstance(data, (str, Path)):
            loaded = _load_chain_simple(str(data))
            if isinstance(loaded, tuple) and len(loaded) == 2:
                data, self._file_metadata = loaded
            else:
                data = loaded

        if isinstance(data, tuple) and len(data) == 2 and isinstance(data[1], dict):
            if 'labels' in data[1] or 'ranges' in data[1]:
                data, self._file_metadata = data

        effective_labels = self._user_labels or self._file_metadata.get('labels')
        effective_ranges = self._user_ranges or self._file_metadata.get('ranges')

        # Derived parameter labels are now stored in file metadata by MCMC.save_results()
        samples = _prepare_getdist_samples(
            data,
            labels=effective_labels,
            ranges=effective_ranges,
        )
        self._samples_list = [samples]
        self._param_names = samples.getParamNames().list()

    def _load_h5_chain(self, chain_name: str) -> Tuple[dict, Dict[str, Tuple[float, float]], Dict[str, str]]:
        """Load samples, ranges and labels from mcmc_chains/{chain_name}.h5

        Returns
        -------
        samples : dict
            Parameter samples {param_name: array}
        ranges : dict
            Parameter bounds {param_name: (min, max)} from MCMC config
        labels : dict
            LaTeX labels {param_name: label} from metadata
        """
        import h5py
        import json

        h5_path = Path('mcmc_chains') / f'{chain_name}.h5'
        if not h5_path.exists():
            raise FileNotFoundError(f"Chain not found: {h5_path}")

        samples = {}
        ranges = {}
        labels = {}
        with h5py.File(h5_path, 'r') as f:
            # Load samples
            if 'samples' in f:
                for param_name in f['samples'].keys():
                    samples[param_name] = np.array(f['samples'][param_name])

            # Load ranges from parameter_config metadata
            if 'metadata' in f and 'parameter_config' in f['metadata']:
                try:
                    config_str = f['metadata']['parameter_config'][()].decode('utf-8')
                    config = json.loads(config_str)
                    if 'parameters' in config:
                        for param_name, param_info in config['parameters'].items():
                            bounds = param_info.get('bounds')
                            if bounds is not None and len(bounds) >= 2:
                                ranges[param_name] = (float(bounds[0]), float(bounds[1]))
                except (json.JSONDecodeError, KeyError, AttributeError):
                    pass  # If parsing fails, ranges will be computed from samples

            # Load LaTeX labels from metadata
            if 'metadata' in f and 'labels' in f['metadata']:
                labels_group = f['metadata']['labels']
                for param_name in labels_group.attrs.keys():
                    labels[param_name] = labels_group.attrs[param_name]

        if not samples:
            raise ValueError(f"No samples found in {h5_path}")

        return samples, ranges, labels

    def _compute_unified_ranges(self, all_samples_data: List) -> Dict[str, Tuple[float, float]]:
        """Compute unified ranges from all chains for consistent plotting."""
        if not all_samples_data:
            return {}

        # Get parameter names from first chain
        if isinstance(all_samples_data[0], dict):
            param_names = list(all_samples_data[0].keys())
        else:
            return {}  # Can't compute ranges for raw arrays without param names

        unified_ranges = {}
        for param in param_names:
            all_values = []
            for chain_data in all_samples_data:
                if isinstance(chain_data, dict) and param in chain_data:
                    all_values.extend(chain_data[param].flatten())

            if all_values:
                arr = np.array(all_values)
                lo, hi = float(np.min(arr)), float(np.max(arr))
                # Add 5% padding
                pad = 0.05 * (hi - lo) if hi > lo else 0.1
                unified_ranges[param] = (lo - pad, hi + pad)

        return unified_ranges

    @classmethod
    def from_fisher(
        cls,
        mean: Union[np.ndarray, List[float]],
        covariance: Union[np.ndarray, List[List[float]]],
        param_names: List[str],
        *,
        nsample: int = 200_000,
        **kwargs,
    ) -> 'Plotter':
        """
        Create Plotter from Fisher matrix results.

        Parameters
        ----------
        mean : array-like
            Mean parameter values
        covariance : array-like
            Covariance matrix
        param_names : list of str
            Parameter names
        nsample : int
            Number of samples to generate (default: 200000)
        **kwargs
            Additional arguments passed to Plotter.__init__

        Returns
        -------
        Plotter
            Plotter instance with generated samples

        Examples
        --------
        >>> plotter = Plotter.from_fisher(
        ...     mean=[67.4, 0.315],
        ...     covariance=[[0.5, 0.01], [0.01, 0.007]],
        ...     param_names=['H0', 'Omega_m']
        ... )
        >>> plotter.corner(filename='forecast.pdf')
        """
        _require_getdist()
        mean_arr = np.asarray(mean, dtype=float)
        cov_arr = np.asarray(covariance, dtype=float)

        if mean_arr.ndim != 1:
            raise ValueError("mean must be a 1-D array")
        if cov_arr.ndim != 2 or cov_arr.shape[0] != cov_arr.shape[1]:
            raise ValueError("covariance must be a square 2-D array")
        if cov_arr.shape[0] != mean_arr.shape[0]:
            raise ValueError("Mean and covariance dimensions do not match")
        if len(param_names) != mean_arr.shape[0]:
            raise ValueError("Number of parameter names must equal the size of the mean vector")

        # Generate LaTeX labels
        latex_labels = _prepare_latex_labels(param_names)

        # Generate samples from Gaussian
        gaussian = GaussianND(mean_arr, cov_arr, names=param_names, labels=latex_labels)
        samples = gaussian.MCSamples(nsample)

        return cls(samples, **kwargs)

    @classmethod
    def from_mcmc(cls, mcmc: 'MCMC', **kwargs) -> 'Plotter':
        """
        Create Plotter from MCMC object.

        Parameters
        ----------
        mcmc : MCMC
            HIcosmo MCMC object with results
        **kwargs
            Additional arguments passed to Plotter.__init__

        Returns
        -------
        Plotter
            Plotter instance
        """
        # Get samples and metadata from MCMC
        samples = mcmc.samples
        labels = mcmc.param_config.get_labels() if hasattr(mcmc, 'param_config') else None
        ranges = mcmc.param_config.get_bounds() if hasattr(mcmc, 'param_config') else None

        return cls(samples, labels=labels, ranges=ranges, **kwargs)

    def corner(
        self,
        params: Optional[Union[List[str], List[int]]] = None,
        *,
        filename: Optional[Union[str, Path]] = None,
        **kwargs,
    ) -> plt.Figure:
        """
        Create corner plot.

        Parameters
        ----------
        params : list, optional
            Parameters to plot. Supports three formats:
            - None: Plot all parameters (default)
            - List of names: ['H0', 'Omega_m', 'rd_h']
            - List of indices (0-based): [0, 1, 2] for first three parameters
        filename : str or Path, optional
            Save filename (auto-saves to results_dir)
        **kwargs
            Additional arguments passed to GetDist

        Returns
        -------
        fig : plt.Figure
            Corner plot figure

        Examples
        --------
        >>> plotter = Plotter('my_chain')
        >>>
        >>> # Plot all parameters
        >>> plotter.corner()
        >>>
        >>> # Plot specific parameters (including derived like rd_h)
        >>> plotter.corner(['Omega_m', 'rd_h'])
        >>>
        >>> # Plot by index (0-based)
        >>> plotter.corner([0, 1, 2])  # First three parameters
        """
        _apply_qijing_style()
        colors = MODERN_COLORS if self.style == 'modern' else CLASSIC_COLORS

        # Select samples for requested parameters
        if params is not None:
            samples_list = self._select_params(params)
        else:
            samples_list = self._samples_list

        # Create plotter
        plotter = plots.get_subplot_plotter(width_inch=8)
        plotter.settings.axes_fontsize = 12
        plotter.settings.lab_fontsize = 14
        plotter.settings.legend_fontsize = 12
        plotter.settings.figure_legend_frame = False

        # Draw corner plot
        if self._is_multichain:
            contour_colors = colors[:len(samples_list)]
            line_colors = colors[:len(samples_list)]

            # Get legend labels
            legend_labels = [s.label for s in samples_list] if any(s.label for s in samples_list) else None

            plotter.triangle_plot(
                samples_list,
                filled=True,
                contour_colors=contour_colors,
                line_args=[{'color': c, 'lw': 2} for c in line_colors],
                legend_labels=legend_labels,
                legend_loc='upper right',
                **kwargs
            )
        else:
            plotter.triangle_plot(
                samples_list,
                filled=True,
                contour_colors=[colors[0]],
                line_args={'color': colors[0], 'lw': 2},
                **kwargs
            )

        _optimize_ticks(plotter.fig)
        _safe_tight_layout(plotter.fig)

        # Save
        if filename:
            self._save_figure(plotter.fig, filename, 'Corner plot')

        return plotter.fig

    def plot_1d(
        self,
        param: Union[str, int],
        *,
        filename: Optional[Union[str, Path]] = None,
        **kwargs,
    ) -> plt.Figure:
        """
        Create 1D marginal distribution plot for a single parameter.

        Parameters
        ----------
        param : str or int
            Parameter name or index to plot
        filename : str or Path, optional
            Save filename
        **kwargs
            Additional arguments passed to GetDist

        Returns
        -------
        fig : plt.Figure
            1D plot figure
        """
        _apply_qijing_style()
        colors = MODERN_COLORS if self.style == 'modern' else CLASSIC_COLORS

        # Resolve parameter name
        if isinstance(param, int):
            param_name = self._param_names[param]
        else:
            param_name = param

        # Create plotter
        plotter = plots.get_single_plotter(width_inch=6)
        plotter.settings.axes_fontsize = 12
        plotter.settings.lab_fontsize = 14
        plotter.settings.legend_fontsize = 9
        plotter.settings.figure_legend_frame = False

        # Get legend labels for multi-chain
        legend_labels = None
        if self._is_multichain and any(s.label for s in self._samples_list):
            legend_labels = [s.label for s in self._samples_list]

        # Set colors
        line_colors = colors[:len(self._samples_list)]

        plotter.plot_1d(
            self._samples_list,
            param_name,
            colors=line_colors,
            **kwargs
        )

        # Add legend manually for multi-chain (no frame, smaller font)
        if legend_labels:
            plotter.add_legend(legend_labels, legend_loc='upper right')
            # Remove legend frame and adjust font size via matplotlib API
            for ax in plotter.fig.axes:
                legend = ax.get_legend()
                if legend:
                    legend.set_frame_on(False)
                    for text in legend.get_texts():
                        text.set_fontsize(9)

        _optimize_ticks(plotter.fig)
        _safe_tight_layout(plotter.fig)

        if filename:
            self._save_figure(plotter.fig, filename, '1D plot')

        return plotter.fig

    def traces(
        self,
        params: Optional[Union[List[str], List[int]]] = None,
        *,
        filename: Optional[Union[str, Path]] = None,
        **kwargs,
    ) -> plt.Figure:
        """
        Create chain trace plots for convergence diagnostics.

        Parameters
        ----------
        params : list, optional
            Parameters to plot (names or indices). If None, plots up to 6.
        filename : str or Path, optional
            Save filename
        **kwargs
            Additional arguments

        Returns
        -------
        fig : plt.Figure
            Trace plot figure
        """
        _apply_qijing_style()
        colors = MODERN_COLORS if self.style == 'modern' else CLASSIC_COLORS

        # Determine parameters to plot
        if params is None:
            n_params = min(6, len(self._param_names))
            selected_params = self._param_names[:n_params]
        else:
            selected_params = self._resolve_param_names(params)

        n_params = len(selected_params)
        fig, axes = plt.subplots(n_params, 1, figsize=(10, 2*n_params))
        if n_params == 1:
            axes = [axes]

        for i, param_name in enumerate(selected_params):
            # Get LaTeX label from first sample
            param_obj = self._samples_list[0].getParamNames().parWithName(param_name)
            latex_label = f"${param_obj.label}$" if param_obj.label else f"${param_name}$"

            for j, samples in enumerate(self._samples_list):
                param_idx = samples.getParamNames().list().index(param_name)
                chain_data = samples.samples[:, param_idx]

                if self._is_multichain:
                    axes[i].plot(chain_data, color=colors[j], alpha=0.8,
                               label=samples.label, lw=1.5)
                else:
                    axes[i].plot(chain_data, color=colors[0], alpha=0.8, lw=1.5)

            axes[i].set_ylabel(latex_label)
            axes[i].grid(True, alpha=0.3)

            if i == 0 and self._is_multichain:
                axes[i].legend(frameon=False)

        axes[-1].set_xlabel('Sample Index')
        _safe_tight_layout(fig)

        if filename:
            self._save_figure(fig, filename, 'Chain traces')

    def marginals(
        self,
        params: Optional[Union[List[str], List[int]]] = None,
        *,
        filename: Optional[Union[str, Path]] = None,
        **kwargs,
    ) -> plt.Figure:
        """
        Create 1D marginal distribution plots.

        Parameters
        ----------
        params : list, optional
            Parameters to plot (names or indices)
        filename : str or Path, optional
            Save filename
        **kwargs
            Additional arguments passed to GetDist

        Returns
        -------
        fig : plt.Figure
            1D marginal plot figure
        """
        _apply_qijing_style()
        colors = MODERN_COLORS if self.style == 'modern' else CLASSIC_COLORS

        # Select samples for requested parameters
        if params is not None:
            samples_list = self._select_params(params)
        else:
            samples_list = self._samples_list

        plotter = plots.get_single_plotter(width_inch=8)
        plotter.settings.axes_fontsize = 12
        plotter.settings.lab_fontsize = 14
        plotter.settings.legend_fontsize = 12
        plotter.settings.figure_legend_frame = False

        # Get legend labels for multi-chain
        legend_labels = None
        if self._is_multichain and any(s.label for s in samples_list):
            legend_labels = [s.label for s in samples_list]

        # Set colors for each chain
        line_colors = colors[:len(samples_list)]

        plotter.plots_1d(
            samples_list,
            colors=line_colors,
            legend_labels=legend_labels,
            **kwargs
        )
        _optimize_ticks(plotter.fig)
        _safe_tight_layout(plotter.fig)

        if filename:
            self._save_figure(plotter.fig, filename, '1D marginals')

        return plotter.fig

    def get_summary(self) -> Dict[str, Dict[str, float]]:
        """
        Get summary statistics for all parameters.

        Returns
        -------
        dict
            Nested dict: {param_name: {'mean': ..., 'std': ..., 'median': ..., 'q16': ..., 'q84': ...}}
        """
        summary = {}

        # Use first chain for single-chain, or aggregate for multi-chain
        samples = self._samples_list[0]

        for i, param_name in enumerate(self._param_names):
            if i < samples.samples.shape[1]:
                data = samples.samples[:, i]
                summary[param_name] = {
                    'mean': float(np.mean(data)),
                    'std': float(np.std(data)),
                    'median': float(np.median(data)),
                    'q16': float(np.percentile(data, 16)),
                    'q84': float(np.percentile(data, 84)),
                }

        return summary

    def _resolve_param_names(self, params: Union[List[str], List[int]]) -> List[str]:
        """
        Convert parameter indices to names if needed.

        Supports:
        - Integer indices (0-based): [0, 1, 2] → first three parameters
        - Parameter names: ['H0', 'Omega_m']
        - Empty list: returns all parameters

        Parameters
        ----------
        params : list of str or int
            Parameter names or indices

        Returns
        -------
        list of str
            Resolved parameter names
        """
        if not params:
            return self._param_names

        if isinstance(params[0], int):
            # 0-based indexing (Python standard)
            resolved = []
            for i in params:
                if i < 0 or i >= len(self._param_names):
                    raise IndexError(
                        f"Parameter index {i} out of range. "
                        f"Available: 0-{len(self._param_names)-1} "
                        f"({', '.join(self._param_names)})"
                    )
                resolved.append(self._param_names[i])
            return resolved
        return list(params)

    def _select_params(self, params: Union[List[str], List[int]]) -> List[MCSamples]:
        """Create MCSamples subset with only requested parameters."""
        selected_param_names = self._resolve_param_names(params)

        result = []
        for samples in self._samples_list:
            # Extract data for selected parameters
            selected_data = []
            selected_labels = []

            for param_name in selected_param_names:
                param_idx = samples.getParamNames().list().index(param_name)
                selected_data.append(samples.samples[:, param_idx])
                selected_labels.append(
                    samples.getParamNames().parWithName(param_name).label
                )

            # Extract ranges for selected parameters
            selected_ranges = {}
            if hasattr(samples, 'ranges') and samples.ranges is not None:
                for param_name in selected_param_names:
                    lower = samples.ranges.getLower(param_name)
                    upper = samples.ranges.getUpper(param_name)
                    if lower is not None and upper is not None:
                        selected_ranges[param_name] = (lower, upper)

            # Create new MCSamples with ranges preserved
            # Set ignore_rows=0.0 because NumPyro already discards warmup
            subset = MCSamples(
                samples=np.column_stack(selected_data),
                names=selected_param_names,
                labels=selected_labels,
                label=getattr(samples, 'label', None),
                ranges=selected_ranges if selected_ranges else None,
                ignore_rows=0.0,  # Avoid double burn-in
            )
            result.append(subset)

        return result

    def _save_figure(self, fig: plt.Figure, filename: Union[str, Path], plot_type: str):
        """Save figure to results directory."""
        save_path = self.results_dir / filename
        if not save_path.suffix:
            save_path = save_path.with_suffix('.pdf')

        save_path.parent.mkdir(parents=True, exist_ok=True)
        fig.savefig(save_path, dpi=300, bbox_inches='tight',
                   facecolor='white', edgecolor='none')
        logger.info("%s saved to: %s", plot_type, save_path)

    def __repr__(self) -> str:
        chain_info = f"{len(self._samples_list)} chains" if self._is_multichain else "single chain"
        return f"Plotter({chain_info}, {len(self._param_names)} parameters)"
